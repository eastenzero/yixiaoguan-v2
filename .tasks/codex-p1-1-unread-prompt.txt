    async def get_unread_summary(db, current_user):
        """For a student, return one entry per conversation with unread_count.

          - sender_type IN (teacher, system)

        Also include last message timestamp + sender_type per conversation (for UI hint).
        Returns UnreadSummaryResponse(items=[...], total_unread=sum(unread_count)).
        Only return conversations belonging to current_user (student).
        Sort items by last_message_at desc; conversations with no messages can be omitted or included with unread_count=0 (your call, document the choice).
        """

    async def mark_conversation_read(db, conv_id, current_user):
        """Set conversations.last_read_at = func.now() for the given conv_id, only if owned by current_user.
        Raises HTTPException 404 if not found / not owned.
        Commits the change.
        """
