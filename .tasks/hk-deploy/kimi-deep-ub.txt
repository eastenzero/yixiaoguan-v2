### 层面 4：ub 上 Dify postgres 里 documents 表是否已经导过大批 KB

ssh ub "docker exec docker-db-1 psql -U postgres -d dify -c \"SELECT d.id, d.name, d.created_at, COUNT(doc.id) AS doc_count FROM datasets d LEFT JOIN documents doc ON doc.dataset_id=d.id GROUP BY d.id, d.name, d.created_at ORDER BY d.created_at DESC LIMIT 10;\" 2>&1 | head -30"
# documents 总数 + 按 dataset 分布
ssh ub "docker exec docker-db-1 psql -U postgres -d dify -c \"SELECT dataset_id, COUNT(*) FROM documents GROUP BY dataset_id ORDER BY 2 DESC LIMIT 10;\" 2>&1 | head"
